Paginação + “linhas por página” + Abas (SenseiSystem)
Objetivo

Adicionar paginação numérica, seletor de linhas/página (10/25/50/100) e abas (ex.: Todas / Ativos / Pendentes) às listas do app (especialmente na tela do aluno).

Funcionamento server-side (com Drizzle), filtro por tenant, ordenação, busca.

URL sincronizada (?page=2&pageSize=25&status=active&q=joao&sort=created_at:desc).

1) Backend — rota paginada
// server/routes/students.ts
import { Router } from "express";
import { authenticateJWT } from "../auth";
import { db } from "../db";
import { and, eq, ilike, sql, desc, asc } from "drizzle-orm";
import { users, students } from "../schema";

export const studentsRouter = Router();

// GET /api/students?page=1&pageSize=25&status=active&q=joao&sort=created_at:desc
studentsRouter.get("/", authenticateJWT, async (req, res) => {
  const tenantId = (req.user as any).tenant_id;

  const page = Math.max(parseInt((req.query.page as string) || "1"), 1);
  const pageSizeRaw = parseInt((req.query.pageSize as string) || "10");
  const pageSize = [10, 25, 50, 100].includes(pageSizeRaw) ? pageSizeRaw : 10;

  const status = (req.query.status as string) || "all"; // all|active|pending|inactive
  const q = (req.query.q as string) || "";
  const sortRaw = (req.query.sort as string) || "created_at:desc";
  const [sortField, sortDir] = sortRaw.split(":");
  const orderBy =
    sortDir === "asc" ? asc(sql.identifier(sortField)) : desc(sql.identifier(sortField));

  const where = and(
    eq(users.tenantId, tenantId),
    eq(users.role, sql.literal("'student'")),
    status === "active" ? eq(users.status, "active") : sql`1=1`,
    status === "pending" ? eq(users.status, "pending") : sql`1=1`,
    status === "inactive" ? eq(users.active, false) : sql`1=1`,
    q ? ilike(sql`(u.first_name || ' ' || u.last_name)`, `%${q}%`) : sql`1=1`,
  );

  // total
  const [{ count }] = await db.execute<{ count: number }>(sql`
    SELECT COUNT(*)::int AS count
    FROM ${students} s
    JOIN ${users} u ON u.id = s.user_id
    WHERE ${where};
  `);

  // page
  const offset = (page - 1) * pageSize;
  const rows = await db.execute(sql`
    SELECT u.id, u.first_name, u.last_name, u.email, u.status, u.active, u.created_at,
           s.belt_level, s.stripes
    FROM ${students} s
    JOIN ${users} u ON u.id = s.user_id
    WHERE ${where}
    ORDER BY ${orderBy}
    LIMIT ${pageSize} OFFSET ${offset};
  `);

  res.json({
    items: rows.rows,
    page,
    pageSize,
    total: count,
    totalPages: Math.max(Math.ceil(count / pageSize), 1),
  });
});


Faça o mesmo padrão para outras listas (aulas, pagamentos…), sempre retornando { items, page, pageSize, total, totalPages }.

2) Front — hook genérico de paginação
// src/hooks/usePaginated.ts
import { useQuery } from "@tanstack/react-query";
import { useSearchParams } from "react-router-dom";

type Opts = {
  key: string;
  endpoint: string; // ex: "/students"
};

export function usePaginated<T>({ key, endpoint }: Opts) {
  const [sp, setSp] = useSearchParams();

  const page = parseInt(sp.get("page") || "1");
  const pageSize = parseInt(sp.get("pageSize") || "10");
  const status = sp.get("status") || "all";
  const q = sp.get("q") || "";
  const sort = sp.get("sort") || "created_at:desc";

  const query = useQuery({
    queryKey: [key, page, pageSize, status, q, sort],
    queryFn: async () => {
      const params = new URLSearchParams({ page: String(page), pageSize: String(pageSize), status, q, sort });
      const res = await api.get(`${endpoint}?${params.toString()}`);
      return res.data as { items: T[]; page: number; pageSize: number; total: number; totalPages: number };
    },
    keepPreviousData: true,
    staleTime: 15_000,
  });

  const setParam = (k: string, v: string) => {
    const next = new URLSearchParams(sp);
    next.set(k, v);
    if (k !== "page") next.set("page", "1"); // reset page on filter changes
    setSp(next, { replace: true });
  };

  return {
    ...query,
    sp,
    setParam,
    page,
    pageSize,
    status,
    q,
    sort,
  };
}

// src/components/ui/Pagination.tsx
function rangeWithDots(total: number, current: number) {
  const pages: (number | "…")[] = [];
  for (let i=1; i<=total; i++) {
    if (i === 1 || i === total || Math.abs(i - current) <= 1) pages.push(i);
  }
  // insert ellipses
  const out:(number|"…")[] = [];
  let prev = 0;
  for (const p of pages) {
    if (typeof p === "number") {
      if (prev && p - prev > 1) out.push("…");
      out.push(p);
      prev = p;
    }
  }
  return out;
}

export function Pagination({ page, totalPages, onPage }:{
  page:number; totalPages:number; onPage:(p:number)=>void;
}) {
  const pages = rangeWithDots(totalPages, page);
  const goto = (p:number) => onPage(Math.max(1, Math.min(totalPages, p)));

  return (
    <div className="flex items-center justify-end gap-1">
      <button className="px-2 py-1 rounded border" onClick={()=>goto(page-1)} disabled={page===1}>‹</button>
      {pages.map((p,i) =>
        p === "…" ? (
          <span key={`d${i}`} className="px-2 text-slate-400">…</span>
        ) : (
          <button
            key={p}
            onClick={()=>goto(p)}
            className={`px-3 py-1 rounded border ${p===page ? "bg-blue-600 text-white border-blue-600" : ""}`}
          >
            {p}
          </button>
        )
      )}
      <button className="px-2 py-1 rounded border" onClick={()=>goto(page+1)} disabled={page===totalPages}>›</button>
    </div>
  );
}

export function Tabs({ value, onChange, items }:{
  value: string;
  onChange: (v:string)=>void;
  items: { value:string; label:string; badge?:number }[];
}) {
  return (
    <div className="flex gap-2 mb-3">
      {items.map(it => (
        <button
          key={it.value}
          onClick={()=>onChange(it.value)}
          className={`px-3 py-1.5 rounded-lg border ${value===it.value ? "bg-slate-900 text-white" : "bg-white"}`}
        >
          {it.label}{typeof it.badge==="number" && <span className="ml-2 text-xs bg-slate-200 rounded px-1">{it.badge}</span>}
        </button>
      ))}
    </div>
  );
}

// src/components/ui/ResultsInfo.tsx
export function ResultsInfo({ page, pageSize, total }:{
  page:number; pageSize:number; total:number;
}) {
  const start = total === 0 ? 0 : (page-1)*pageSize + 1;
  const end = Math.min(page*pageSize, total);
  return <div className="text-sm text-slate-600">Mostrando {start}–{end} de {total} resultados</div>;
}

4) Tela do Aluno (listagem) — exemplo de uso
// src/pages/alunos/ListaAlunos.tsx
import { usePaginated } from "@/hooks/usePaginated";
import { Pagination } from "@/components/ui/Pagination";
import { PageSizeSelect } from "@/components/ui/PageSizeSelect";
import { Tabs } from "@/components/ui/Tabs";
import { ResultsInfo } from "@/components/ui/ResultsInfo";

export default function ListaAlunos() {
  const { data, isFetching, page, pageSize, setParam, status, q } =
    usePaginated<{ id:number; first_name:string; last_name:string; email:string; status:string; active:boolean; belt_level:string; stripes:number; created_at:string; }>({
      key: "students",
      endpoint: "/students",
    });

  return (
    <div className="space-y-3">
      {/* Filtros superiores */}
      <div className="flex items-center justify-between gap-3 flex-wrap">
        <Tabs
          value={status}
          onChange={(v)=>setParam("status", v)}
          items={[
            { value: "all",     label: "Todas" },
            { value: "active",  label: "Ativos" },
            { value: "pending", label: "Pendentes" },
            { value: "inactive",label: "Inativos" },
          ]}
        />
        <div className="flex items-center gap-3">
          <input
            className="border rounded-md px-3 py-1.5 w-56"
            placeholder="Buscar por nome/email"
            defaultValue={q}
            onKeyDown={(e) => {
              if (e.key === "Enter") setParam("q", (e.target as HTMLInputElement).value);
            }}
          />
          <PageSizeSelect value={pageSize} onChange={(n)=>setParam("pageSize", String(n))} />
        </div>
      </div>

      {/* Tabela */}
      <div className="overflow-x-auto rounded-xl border bg-white">
        <table className="min-w-full text-sm">
          <thead className="bg-slate-50">
            <tr className="[&>th]:px-3 [&>th]:py-2 text-left">
              <th>Aluno</th>
              <th>E-mail</th>
              <th>Status</th>
              <th>Faixa</th>
              <th className="text-right pr-3">Ações</th>
            </tr>
          </thead>
          <tbody className="[&>tr>td]:px-3 [&>tr>td]:py-2">
            {!data?.items.length && !isFetching && (
              <tr><td colSpan={5} className="text-center py-8 text-slate-500">Nenhum resultado</td></tr>
            )}
            {data?.items.map((aluno) => (
              <tr key={aluno.id} className="border-t">
                <td>{aluno.first_name} {aluno.last_name}</td>
                <td>{aluno.email}</td>
                <td>{aluno.active ? "Ativo" : aluno.status}</td>
                <td>{aluno.belt_level} ({aluno.stripes} listras)</td>
                <td className="text-right">
                  <a className="text-blue-600 hover:underline" href={`/alunos/${aluno.id}`}>Abrir</a>
                </td>
              </tr>
            ))}
          </tbody>
        </table>
      </div>

      {/* Footer da paginação */}
      <div className="flex items-center justify-between">
        <ResultsInfo page={data?.page ?? 1} pageSize={data?.pageSize ?? 10} total={data?.total ?? 0} />
        <Pagination
          page={data?.page ?? 1}
          totalPages={data?.totalPages ?? 1}
          onPage={(p)=>setParam("page", String(p))}
        />
      </div>
    </div>
  );
}

Critérios de aceite

 Listas exibem Mostrando X–Y de N resultados e paginação (‹ 1 … N ›).

 Linhas por página com 10/25/50/100 (recarrega mantendo filtros).

 Abas mudam status (URL atualiza; página volta para 1).

 Busca por Enter (mantém filtros/ordenar).

 Tudo server-side com tenant_id aplicado.

 Acessível: botões com :focus, labels e aria quando necessário.

2) Utilitário para avaliar “risco” no questionário
// server/utils/healthRisk.ts
export type HealthAnswer = { key: string; question: string; value: boolean | string | null };

const RISK_KEYS = new Set([
  "heartCondition",
  "chestPain",
  "fainting",
  "doctorRestriction",
  "recentSurgery",
  "medicationAffectsExercise",
  "pregnancy",
]);

export function isRisky(answers: HealthAnswer[]): boolean {
  for (const a of answers) {
    if (RISK_KEYS.has(a.key) && (a.value === true || String(a.value).toLowerCase() === "sim")) {
      return true;
    }
  }
  return false;
}

3) Geração do PDF do questionário (PDFKit)
// server/services/healthQuestionnairePdf.ts
import PDFDocument from "pdfkit";
import fs from "fs";
import path from "path";
import type { HealthAnswer } from "../utils/healthRisk";

export async function generateHealthPdf(
  studentFullName: string,
  answers: HealthAnswer[],
  submittedAt: Date,
  outDir: string,
  studentId: number
): Promise<{ filePath: string; size: number; fileName: string }> {
  fs.mkdirSync(outDir, { recursive: true });

  const fileName = `questionario-saude_${studentId}_${submittedAt.getTime()}.pdf`;
  const filePath = path.join(outDir, fileName);

  return new Promise((resolve, reject) => {
    const doc = new PDFDocument({ size: "A4", margin: 50 });
    const stream = fs.createWriteStream(filePath);
    doc.pipe(stream);

    doc.fontSize(18).text("Questionário de Saúde", { align: "center" });
    doc.moveDown(0.5);
    doc.fontSize(12).text(`Aluno: ${studentFullName}`);
    doc.text(`Data/Hora de preenchimento: ${submittedAt.toLocaleString("pt-BR")}`);
    doc.moveDown();

    answers.forEach((a, i) => {
      doc.font("Helvetica-Bold").text(`${i + 1}. ${a.question}`);
      doc.font("Helvetica").text(`Resposta: ${a.value === true ? "Sim" : a.value === false ? "Não" : a.value ?? "-"}`);
      doc.moveDown(0.5);
    });

    doc.end();
    stream.on("finish", () => {
      const stats = fs.statSync(filePath);
      resolve({ filePath, size: stats.size, fileName });
    });
    stream.on("error", reject);
  });
}

4) Service: salvar questionário, anexar PDF, e marcar exigência de atestado
// server/services/healthQuestionnaireService.ts
import path from "path";
import { prisma } from "../prisma";
import { isRisky, HealthAnswer } from "../utils/healthRisk";
import { generateHealthPdf } from "./healthQuestionnairePdf";

const STORAGE_ROOT = process.env.STORAGE_ROOT || path.join(process.cwd(), "storage");

export async function saveHealthQuestionnaire(studentId: number, answers: HealthAnswer[]) {
  const risky = isRisky(answers);
  const submittedAt = new Date();

  // upsert do questionário
  await prisma.healthQuestionnaire.upsert({
    where: { studentId },
    create: { studentId, answersJson: answers, risky, submittedAt },
    update: { answersJson: answers, risky, submittedAt },
  });

  // nome completo do aluno
  const s = await prisma.student.findUnique({ where: { id: studentId }, include: { user: true } });
  if (!s?.user) throw new Error("Aluno não encontrado para gerar PDF.");

  // gerar PDF e registrar como Document
  const outDir = path.join(STORAGE_ROOT, "students", String(studentId), "health");
  const { filePath, size, fileName } = await generateHealthPdf(
    `${s.user.firstName} ${s.user.lastName}`.trim(),
    answers,
    submittedAt,
    outDir,
    studentId
  );

  // atualizar caminho no questionário
  await prisma.healthQuestionnaire.update({
    where: { studentId },
    data: { pdfPath: filePath, pdfSize: size },
  });

  // registrar/atualizar document kind=HEALTH_FORM_PDF
  const exists = await prisma.document.findFirst({
    where: { studentId, kind: "HEALTH_FORM_PDF" },
  });

  if (exists) {
    await prisma.document.update({
      where: { id: exists.id },
      data: { name: fileName, mime: "application/pdf", size, path: filePath, uploadedAt: submittedAt },
    });
  } else {
    await prisma.document.create({
      data: {
        studentId,
        kind: "HEALTH_FORM_PDF",
        name: fileName,
        mime: "application/pdf",
        size,
        path: filePath,
      },
    });
  }

  // marcar necessidade de atestado
  await prisma.student.update({
    where: { id: studentId },
    data: {
      requiresMedicalCertificate: risky,
      medicalCertificateStatus: risky ? "PENDING" : "WAIVED",
    },
  });

  return { risky, submittedAt, fileName, size };
}

// server/services/uploadService.ts
import path from "path";
import fs from "fs";

const STORAGE_ROOT = process.env.STORAGE_ROOT || path.join(process.cwd(), "storage");

export function ensureDir(p: string) { fs.mkdirSync(p, { recursive: true }); }

export function destinationForStudentKind(studentId: number, kind: string) {
  const dir = path.join(STORAGE_ROOT, "students", String(studentId), kind.toLowerCase());
  ensureDir(dir);
  return dir;
}

// server/routes.uploads.ts (multer config simplificado)
import multer from "multer";
import path from "path";
import { destinationForStudentKind } from "./services/uploadService";

const storage = multer.diskStorage({
  destination: (req, _file, cb) => {
    const studentId = Number(req.params.id);
    const kind = String(req.params.kind || "misc");
    const dir = destinationForStudentKind(studentId, kind);
    cb(null, dir);
  },
  filename: (_req, file, cb) => {
    const name = `${Date.now()}_${file.originalname.replace(/\s+/g, "_")}`;
    cb(null, name);
  },
});

export const upload = multer({
  storage,
  limits: { fileSize: 10 * 1024 * 1024 }, // 10MB
  fileFilter: (_req, file, cb) => {
    const ok = ["application/pdf", "image/png", "image/jpeg"].includes(file.mimetype);
    cb(ok ? null : new Error("Tipo de arquivo inválido"), ok);
  },
});

import { saveHealthQuestionnaire } from "./services/healthQuestionnaireService";

app.post("/api/students/:id/health-questionnaire", async (req: Request, res: Response) => {

    const id = Number(req.params.id);
    const answers = req.body?.answers || [];
    const result = await saveHealthQuestionnaire(id, answers);
    return res.json({ success: true, ...result });

    return res.status(500).json({ success: false, message: "Erro ao salvar o questionário." });
  }

// GET: baixar PDF do questionário
app.get("/api/students/:id/health-questionnaire/pdf", async (req: Request, res: Response) => {
  const id = Number(req.params.id);
  const q = await prisma.healthQuestionnaire.findUnique({ where: { studentId: id } });
  if (!q?.pdfPath || !fs.existsSync(q.pdfPath)) return res.status(404).send("PDF não encontrado");
  res.setHeader("Content-Type", "application/pdf");
  res.setHeader("Content-Disposition", `attachment; filename="questionario-saude_${id}.pdf"`);
  fs.createReadStream(q.pdfPath).pipe(res);
});

// POST upload de documento (ex.: atestado)
app.post("/api/students/:id/documents/:kind", upload.single("file"), async (req: Request, res: Response) => {
  try {
    const id = Number(req.params.id);
    const kind = String(req.params.kind); // "MEDICAL_CERT" | "RG_CNH" | ...
    const file = req.file!;
    const doc = await prisma.document.create({
      data: {
        studentId: id,
        kind,
        name: file.originalname,
        mime: file.mimetype,
        size: file.size,
        path: file.path,
      },
    });

    if (kind === "MEDICAL_CERT") {
      await prisma.student.update({
        where: { id },
        data: { medicalCertificateStatus: "UPLOADED" },
      });
    }

    return res.json({ success: true, document: doc });
  } catch (e: any) {
    console.error("upload document error", e);
    return res.status(500).json({ success: false, message: "Erro ao enviar documento." });
  }
});

// frontend/src/services/api/health.ts
import axios from "axios";

export type HealthAnswer = { key: string; question: string; value: boolean | string | null };

export async function submitHealthQuestionnaire(studentId: number, answers: HealthAnswer[]) {
  const { data } = await axios.post(`/api/students/${studentId}/health-questionnaire`, { answers });
  return data as { success: boolean; risky: boolean; submittedAt: string };
}

export function downloadHealthPdf(studentId: number) {
  window.location.href = `/api/students/${studentId}/health-questionnaire/pdf`;
}

export async function listDocuments(studentId: number) {
  const { data } = await axios.get(`/api/students/${studentId}/documents`);
  return data as Array<{ id: number; kind: string; name: string; size: number }>;
}


Componente do Questionário (envio + feedback + exigência do atestado)

// frontend/src/components/onboarding/HealthQuestionnaire.tsx
import { useState } from "react";
import { submitHealthQuestionnaire, downloadHealthPdf } from "@/services/api/health";
import { Button } from "@/components/ui/button";
import { useToast } from "@/components/ui/use-toast";

export default function HealthQuestionnaire({ studentId, questions }: { studentId: number; questions: Array<{ key: string; label: string; type: "yesno" | "text" }>}) {
  const { toast } = useToast();
  const [answers, setAnswers] = useState<Record<string, any>>({});
  const [saving, setSaving] = useState(false);
  const [submitted, setSubmitted] = useState(false);
  const [risky, setRisky] = useState(false);

  function setAnswer(key: string, value: any) {
    setAnswers((s) => ({ ...s, [key]: value }));
  }

  async function onSubmit() {
    setSaving(true);
    try {
      const payload = questions.map(q => ({ key: q.key, question: q.label, value: answers[q.key] ?? null }));
      const res = await submitHealthQuestionnaire(studentId, payload);
      setSubmitted(true);
      setRisky(res.risky);
      toast({ title: "Questionário salvo", description: res.risky ? "Identificamos respostas que exigem atestado médico." : "Tudo certo!" });
    } catch (e: any) {
      toast({ title: "Erro", description: "Não foi possível salvar o questionário.", variant: "destructive" });
    } finally {
      setSaving(false);
    }
  }

  return (
    <div className="space-y-4">
      {questions.map((q) => (
        <div key={q.key}>
          <p className="text-sm font-medium">{q.label}</p>
          {q.type === "yesno" ? (
            <div className="flex gap-2 mt-1">
              <Button type="button" variant={answers[q.key] === true ? "default" : "outline"} onClick={() => setAnswer(q.key, true)}>Sim</Button>
              <Button type="button" variant={answers[q.key] === false ? "default" : "outline"} onClick={() => setAnswer(q.key, false)}>Não</Button>
            </div>
          ) : (
            <input className="border rounded px-3 py-2 w-full mt-1" onChange={(e)=> setAnswer(q.key, e.target.value)} />
          )}
        </div>
      ))}

      <div className="flex items-center gap-2">
        <Button onClick={onSubmit} disabled={saving}>{saving ? "Salvando..." : "Salvar Questionário"}</Button>
        {submitted && (
          <Button type="button" variant="secondary" onClick={()=>downloadHealthPdf(studentId)}>
            Baixar PDF
          </Button>
        )}
      </div>

      {submitted && risky && (
        <div className="rounded-md bg-amber-50 border border-amber-200 p-3 text-amber-800">
          Foi identificado que você **precisa anexar um Atestado Médico**. O status ficará <b>Pendente</b> até o envio.
        </div>
      )}
    </div>
  );
}


Tela de Documentos (com botão Baixar do PDF e upload do atestado obrigatório)

// frontend/src/components/onboarding/DocumentsBlock.tsx
import { useEffect, useState } from "react";
import axios from "axios";
import { listDocuments, downloadHealthPdf } from "@/services/api/health";
import { Button } from "@/components/ui/button";
import { useToast } from "@/components/ui/use-toast";

export default function DocumentsBlock({ studentId, requiresMedical }: { studentId: number; requiresMedical: boolean }) {
  const { toast } = useToast();
  const [docs, setDocs] = useState<any[]>([]);
  const [uploading, setUploading] = useState(false);

  async function refresh() {
    const d = await listDocuments(studentId);
    setDocs(d);
  }
  useEffect(() => { refresh(); }, []);

  async function uploadMedicalCert(file: File) {
    setUploading(true);
    try {
      const form = new FormData();
      form.append("file", file);
      await axios.post(`/api/students/${studentId}/documents/MEDICAL_CERT`, form, { headers: { "Content-Type": "multipart/form-data" } });
      toast({ title: "Atestado enviado" });
      refresh();
    } catch {
      toast({ title: "Erro ao enviar", variant: "destructive" });
    } finally {
      setUploading(false);
    }
  }

  return (
    <div className="space-y-4">
      <div className="rounded border p-3">
        <div className="flex items-center justify-between">
          <div>
            <p className="font-medium">Questionário de Saúde</p>
            <p className="text-sm text-muted-foreground">PDF com as respostas enviadas.</p>
          </div>
          <Button onClick={() => downloadHealthPdf(studentId)}>Baixar</Button>
        </div>
      </div>

      <div className="rounded border p-3">
        <div className="flex items-center justify-between mb-2">
          <div>
            <p className="font-medium">Atestado Médico {requiresMedical ? "(Obrigatório)" : "(Opcional)"}</p>
            <p className="text-sm text-muted-foreground">PDF, JPG ou PNG até 10MB.</p>
          </div>
          {requiresMedical && !docs.some(d => d.kind === "MEDICAL_CERT") && (
            <span className="text-xs px-2 py-1 rounded bg-slate-200">Pendente</span>
          )}
        </div>

        <label className="block border-2 border-dashed rounded p-6 text-center cursor-pointer hover:bg-muted">
          <input type="file" className="hidden" accept="application/pdf,image/png,image/jpeg"
            onChange={(e)=> e.target.files?.[0] && uploadMedicalCert(e.target.files[0])} />
          {uploading ? "Enviando..." : "Clique para enviar ou arraste o arquivo"}
        </label>

        {/* lista de anexos */}
        <div className="mt-3 space-y-2">
          {docs.filter(d => d.kind === "MEDICAL_CERT").map((d) => (
            <div key={d.id} className="flex items-center justify-between rounded bg-green-50 border border-green-200 px-3 py-2">
              <div>
                <p className="text-sm">{d.name}</p>
                <p className="text-xs text-muted-foreground">{(d.size/1024).toFixed(1)} KB</p>
              </div>
              <Button variant="secondary" onClick={() => (window.location.href = `/api/documents/${d.id}/download`)}>Baixar</Button>
            </div>
          ))}
        </div>
      </div>
    </div>
  );
}

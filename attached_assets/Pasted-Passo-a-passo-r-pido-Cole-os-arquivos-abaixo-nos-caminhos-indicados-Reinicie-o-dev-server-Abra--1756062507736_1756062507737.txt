// frontend/src/lib/formatters.ts
export const onlyDigits = (v?: string | null) => (v || "").replace(/\D+/g, "");

export function formatCPF(v?: string | null) {
  const s = onlyDigits(v);
  return s
    .slice(0, 11)
    .replace(/^(\d{3})(\d)/, "$1.$2")
    .replace(/^(\d{3})\.(\d{3})(\d)/, "$1.$2.$3")
    .replace(/\.(\d{3})(\d)/, ".$1-$2");
}
export function unformatCPF(v?: string | null) {
  return onlyDigits(v).slice(0, 11);
}

export function formatRG(v?: string | null) {
  const s = onlyDigits(v).slice(0, 9);
  if (s.length <= 2) return s;
  if (s.length <= 5) return s.replace(/^(\d{2})(\d+)/, "$1.$2");
  if (s.length <= 8) return s.replace(/^(\d{2})(\d{3})(\d+)/, "$1.$2.$3");
  return s.replace(/^(\d{2})(\d{3})(\d{3})(\d)/, "$1.$2.$3-$4");
}
export const unformatRG = (v?: string | null) => onlyDigits(v).slice(0, 9);

export function formatPhone(v?: string | null) {
  const s = onlyDigits(v).slice(0, 11);
  if (s.length <= 10) {
    return s.replace(/^(\d{2})(\d{4})(\d{0,4})$/, "($1) $2-$3").trim();
  }
  return s.replace(/^(\d{2})(\d{5})(\d{0,4})$/, "($1) $2-$3").trim();
}
export const unformatPhone = (v?: string | null) => onlyDigits(v).slice(0, 11);

export function formatCEP(v?: string | null) {
  return onlyDigits(v).slice(0, 8).replace(/^(\d{5})(\d{0,3})$/, "$1-$2").trim();
}
export const unformatCEP = (v?: string | null) => onlyDigits(v).slice(0, 8);

export function toDisplayDate(iso?: string | null) {
  if (!iso) return "";
  const d = new Date(iso);
  if (Number.isNaN(d.getTime())) return "";
  const dd = String(d.getDate()).padStart(2, "0");
  const mm = String(d.getMonth() + 1).padStart(2, "0");
  const yyyy = d.getFullYear();
  return `${dd}/${mm}/${yyyy}`;
}
export function toISODate(display?: string | null) {
  if (!display) return null;
  const m = display.match(/^(\d{2})\/(\d{2})\/(\d{4})$/);
  if (!m) return null;
  const [_, dd, mm, yyyy] = m;
  return `${yyyy}-${mm}-${dd}`;
}

  id: number;
  firstName: string;

  birthDate: string | null;
  cpf: string | null;
  rg: string | null;
  contact: { phones: string[]; email: string | null };

  const { data } = await axios.get(`/api/students/${id}?include=all`);
  return data as StudentEditDTO;
}

  const { data } = await axios.patch(`/api/students/${id}`, payload);
  return data as StudentEditDTO;

// server/services/studentsService.ts
import { prisma } from "../prisma"; // ajuste se necessário
import { StudentEditDTO } from "../dto/StudentEditDTO";

const nd = (v?: string | null) => (v == null ? null : v.trim() || null);

export async function getStudentEditDTO(id: number): Promise<StudentEditDTO | null> {
  const s = await prisma.student.findUnique({
    where: { id },
    include: {
      user: true,
      contact: true,
      address: true,
      health: true,
      graduation: true,
      guardian: true,
    },
  });
  if (!s || !s.user) return null;

  return {
    id: s.id,
    firstName: s.user.firstName,
    lastName: s.user.lastName,
    birthDate: s.birthDate ? s.birthDate.toISOString().slice(0, 10) : null,
    cpf: s.cpf || null,
    rg: s.rg || null,
    contact: {
      phones: s.contact?.phones || [],
      email: s.contact?.email || null,
    },
    address: {
      street: s.address?.street || null,
      number: s.address?.number || null,
      district: s.address?.district || null,
      city: s.address?.city || null,
      state: s.address?.state || null,
      zip: s.address?.zip || null,
    },
    health: { notes: s.health?.notes || null },
    graduation: {
      beltLevel: s.graduation?.beltLevel || null,
      graduationDate: s.graduation?.graduationDate
        ? s.graduation.graduationDate.toISOString().slice(0, 10)
        : null,
    },
    guardian: {
      name: s.guardian?.name || null,
      relation: s.guardian?.relation || null,
      phone: s.guardian?.phone || null,
      email: s.guardian?.email || null,
    },
  };
}

export async function patchStudentFromDTO(id: number, dto: StudentEditDTO): Promise<StudentEditDTO> {
  // transação para manter consistência
  await prisma.$transaction(async (tx) => {
    await tx.user.update({
      where: { id: (await tx.student.findUniqueOrThrow({ where: { id } })).userId },
      data: { firstName: dto.firstName, lastName: dto.lastName },
    });

    await tx.student.update({
      where: { id },
      data: {
        birthDate: dto.birthDate ? new Date(dto.birthDate) : null,
        cpf: dto.cpf || null,
        rg: dto.rg || null,
      },
    });

    // contato
    const hasContact = await tx.contact.findUnique({ where: { studentId: id } });
    if (hasContact) {
      await tx.contact.update({
        where: { studentId: id },
        data: { phones: dto.contact.phones || [], email: nd(dto.contact.email) },
      });
    } else {
      await tx.contact.create({
        data: { studentId: id, phones: dto.contact.phones || [], email: nd(dto.contact.email) },
      });
    }

    // endereço
    const hasAddress = await tx.address.findUnique({ where: { studentId: id } });
    const addrData = {
      street: nd(dto.address.street),
      number: nd(dto.address.number),
      district: nd(dto.address.district),
      city: nd(dto.address.city),
      state: nd(dto.address.state),
      zip: nd(dto.address.zip),
    };
    if (hasAddress) {
      await tx.address.update({ where: { studentId: id }, data: addrData });
    } else {
      await tx.address.create({ data: { studentId: id, ...addrData } });
    }

    // saúde
    const hasHealth = await tx.health.findUnique({ where: { studentId: id } });
    if (hasHealth) {
      await tx.health.update({ where: { studentId: id }, data: { notes: nd(dto.health.notes) } });
    } else {
      await tx.health.create({ data: { studentId: id, notes: nd(dto.health.notes) } });
    }

    // graduação
    const hasGrad = await tx.graduation.findUnique({ where: { studentId: id } });
    const gradData = {
      beltLevel: nd(dto.graduation.beltLevel),
      graduationDate: dto.graduation.graduationDate ? new Date(dto.graduation.graduationDate) : null,
    };
    if (hasGrad) {
      await tx.graduation.update({ where: { studentId: id }, data: gradData });
    } else {
      await tx.graduation.create({ data: { studentId: id, ...gradData } });
    }

    // responsável
    const hasGuardian = await tx.guardian.findUnique({ where: { studentId: id } });
    const gData = {
      name: nd(dto.guardian.name),
      relation: nd(dto.guardian.relation),
      phone: nd(dto.guardian.phone),
      email: nd(dto.guardian.email),
    };
    if (hasGuardian) {
      await tx.guardian.update({ where: { studentId: id }, data: gData });
    } else {
      await tx.guardian.create({ data: { studentId: id, ...gData } });
    }
  });

  // retornar DTO atualizado
  const updated = await getStudentEditDTO(id);
  if (!updated) throw new Error("Falha ao recarregar aluno.");
  return updated;
}


Dica: se seus nomes de tabelas diferirem, mapeie conforme o seu schema. O importante é manter um DTO único e estável.

3) Rotas (GET/PATCH)

server/routes.ts (apenas o trecho das rotas do aluno)

// server/routes.ts (trechos relevantes)
import type { Request, Response } from "express";
import { getStudentEditDTO, patchStudentFromDTO } from "./services/studentsService";
import { StudentEditDTO } from "./dto/StudentEditDTO";

app.get("/api/students/:id", async (req: Request, res: Response) => {
  try {
    const id = Number(req.params.id);
    const dto = await getStudentEditDTO(id);
    if (!dto) return res.status(404).json({ message: "Aluno não encontrado" });
    return res.json(dto);
  } catch (e: any) {
    console.error("GET /api/students/:id error", e);
    return res.status(500).json({ message: "Erro ao carregar aluno." });
  }
});

app.patch("/api/students/:id", async (req: Request, res: Response) => {
  try {
    const id = Number(req.params.id);
    const body: StudentEditDTO = req.body;
    if (id !== body.id) return res.status(400).json({ message: "ID inconsistente." });
    const updated = await patchStudentFromDTO(id, body);
    return res.json(updated);
  } catch (e: any) {
    console.error("PATCH /api/students/:id error", e);
    return res.status(500).json({ message: "Erro ao salvar dados do aluno.", detail: e?.message });
  }
});

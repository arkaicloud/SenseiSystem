Onboarding (anti‑circular)

function onSubmit(e){ 
  e.preventDefault();
  const fd = new FormData(e.currentTarget);
  const payload = {
    name: String(fd.get('name')||''),
    email: String(fd.get('email')||''),
    planId: String(fd.get('planId')||''),
    plan_price_cents: brlToCents(String(fd.get('planPrice')||'')), // se aplicável
  };
  return api.students.onboard(payload);
}

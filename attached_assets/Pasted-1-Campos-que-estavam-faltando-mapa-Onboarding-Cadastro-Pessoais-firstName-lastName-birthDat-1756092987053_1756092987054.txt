export async function getStudentById(id: number) {
  const { data } = await axios.get(`/api/students/${id}?include=all`, { headers: { "Cache-Control": "no-store" }});

}

export async function updateStudent(id: number, payload: StudentEditDTO) {
  const { data } = await axios.patch(`/api/students/${id}`, payload);

Pontos‑chave da correção “abre aluno errado”:

Forçar re-render do dialog por chave key={studentId}.

Abortar fetch anterior ao trocar de aluno (race condition).

no-store no GET (evita 304 com payload de outro aluno em proxies agressivos).

useEffect dependente de studentId e open.

form.reset() apenas depois que a resposta daquele id chegar.

import { prisma } from "../prisma";
import { StudentEditDTO } from "../dto/StudentEditDTO";

const nd = (v?: string | null) => (v == null ? null : v.trim() || null);
const onlyDigits = (v?: string | null) => (v || "").replace(/\D+/g, "");

export async function getStudentEditDTO(id: number): Promise<StudentEditDTO | null> {
  const s = await prisma.student.findUnique({
    where: { id },
    include: {
      user: true,
      contact: true,
      emergency: true,
      address: true,
      financialResponsible: true,
      billingPreference: true,
      health: true,
      graduation: true,
    },
  });
  if (!s || !s.user) return null;

  return {
    id: s.id,
    firstName: s.user.firstName,
    lastName: s.user.lastName,
    birthDate: s.birthDate ? s.birthDate.toISOString().slice(0,10) : null,
    cpf: s.cpf || null,
    rg: s.rg || null,
    sex: (s.sex as any) ?? null,
    contact: {
      email: s.contact?.email || null,
      phone: s.contact?.phone || null,
    },
    emergency: {
      name: s.emergency?.name || null,
      phone: s.emergency?.phone || null,
    },
    financialResponsible: { relation: s.financialResponsible?.relation || null },
    billing: {
      planId: s.billingPreference?.planId ?? null,
      preferredDueDay: s.billingPreference?.preferredDueDay ?? null,
    },
    address: {
      zip: s.address?.zip || null,
      street: s.address?.street || null,
      number: s.address?.number || null,
      complement: s.address?.complement || null,
      district: s.address?.district || null,
      city: s.address?.city || null,
      state: s.address?.state || null,
    },
    health: { notes: s.health?.notes || null },
    graduation: {
      beltLevel: s.graduation?.beltLevel || null,
      graduationDate: s.graduation?.graduationDate ? s.graduation.graduationDate.toISOString().slice(0,10) : null,
    },
  };
}

export async function patchStudentFromDTO(id: number, dto: StudentEditDTO): Promise<StudentEditDTO> {
  await prisma.$transaction(async (tx) => {
    const st = await tx.student.findUniqueOrThrow({ where: { id }});

    await tx.user.update({
      where: { id: st.userId },
      data: { firstName: nd(dto.firstName), lastName: nd(dto.lastName) },
    });

    await tx.student.update({
      where: { id },
      data: {
        birthDate: dto.birthDate ? new Date(dto.birthDate) : null,
        cpf: onlyDigits(dto.cpf),
        rg: onlyDigits(dto.rg),
        sex: dto.sex ?? null,
      },
    });

    // contato
    const c = await tx.contact.findUnique({ where: { studentId: id }});
    const cData = { email: nd(dto.contact.email), phone: onlyDigits(dto.contact.phone) || null, studentId: id };
    c ? await tx.contact.update({ where: { studentId: id }, data: cData }) : await tx.contact.create({ data: cData });

    // emergência
    const em = await tx.emergency.findUnique({ where: { studentId: id }});
    const emData = { name: nd(dto.emergency.name), phone: onlyDigits(dto.emergency.phone) || null, studentId: id };
    em ? await tx.emergency.update({ where: { studentId: id }, data: emData }) : await tx.emergency.create({ data: emData });

    // responsável financeiro
    const fr = await tx.financialResponsible.findUnique({ where: { studentId: id }});
    const frData = { relation: nd(dto.financialResponsible.relation), studentId: id };
    fr ? await tx.financialResponsible.update({ where: { studentId: id }, data: frData }) : await tx.financialResponsible.create({ data: frData });

    // preferência de cobrança
    const bp = await tx.billingPreference.findUnique({ where: { studentId: id }});
    const bpData = { planId: dto.billing.planId, preferredDueDay: dto.billing.preferredDueDay, studentId: id };
    bp ? await tx.billingPreference.update({ where: { studentId: id }, data: bpData }) : await tx.billingPreference.create({ data: bpData });

    // endereço
    const ad = await tx.address.findUnique({ where: { studentId: id }});
    const adData = {
      zip: onlyDigits(dto.address.zip) || null,
      street: nd(dto.address.street), number: nd(dto.address.number),
      complement: nd(dto.address.complement), district: nd(dto.address.district),
      city: nd(dto.address.city), state: nd(dto.address.state), studentId: id
    };
    ad ? await tx.address.update({ where: { studentId: id }, data: adData }) : await tx.address.create({ data: adData });

    // saúde
    const he = await tx.health.findUnique({ where: { studentId: id }});
    he ? await tx.health.update({ where: { studentId: id }, data: { notes: nd(dto.health.notes) }})
       : await tx.health.create({ data: { notes: nd(dto.health.notes), studentId: id }});

    // graduação
    const gr = await tx.graduation.findUnique({ where: { studentId: id }});
    const grData = { beltLevel: nd(dto.graduation.beltLevel), graduationDate: dto.graduation.graduationDate ? new Date(dto.graduation.graduationDate) : null, studentId: id };
    gr ? await tx.graduation.update({ where: { studentId: id }, data: grData })
       : await tx.graduation.create({ data: grData });
  });

  const updated = await getStudentEditDTO(id);
  if (!updated) throw new Error("Falha ao recarregar aluno.");
  return updated;
}

import type { Request, Response } from "express";
import { getStudentEditDTO, patchStudentFromDTO } from "./services/studentsService";
import { StudentEditDTO } from "./dto/StudentEditDTO";

// GET
app.get("/api/students/:id", async (req: Request, res: Response) => {
  try {
    const id = Number(req.params.id);
    const dto = await getStudentEditDTO(id);
    res.setHeader("Cache-Control", "no-store");
    if (!dto) return res.status(404).json({ message: "Aluno não encontrado" });
    return res.json(dto);
  } catch (e: any) {
    console.error("GET /api/students/:id error", e);
    return res.status(500).json({ message: "Erro ao carregar aluno." });
  }
});

// PATCH
app.patch("/api/students/:id", async (req: Request, res: Response) => {
  try {
    const id = Number(req.params.id);
    const body: StudentEditDTO = req.body;
    if (id !== body.id) return res.status(400).json({ message: "ID inconsistente." });
    const updated = await patchStudentFromDTO(id, body);
    return res.json(updated);
  } catch (e: any) {
    console.error("PATCH /api/students/:id error", e);
    return res.status(500).json({ message: "Erro ao salvar dados do aluno.", detail: e?.message });
  }
});

// STATE
let curYear = 2025, curMonth = 5;
let selDate = null, selClass = null;
let students = [], selected = new Set(), filter = 'all', searchQ = '';

// CALENDAR
function renderCal() {
  document.getElementById('cal-month').textContent = MONTHS[curMonth] + ' ' + curYear;
  const c = document.getElementById('cal-days');
  c.innerHTML = '';
  const today = new Date();
  const first = new Date(curYear, curMonth, 1).getDay();
  const days = new Date(curYear, curMonth+1, 0).getDate();

  for (let i=0;i<first;i++) {
    const e=document.createElement('div'); e.className='cal-day empty'; c.appendChild(e);
  }
  for (let d=1;d<=days;d++) {
    const ds = `${curYear}-${String(curMonth+1).padStart(2,'0')}-${String(d).padStart(2,'0')}`;
    const e = document.createElement('div');
    e.className = 'cal-day';
    e.textContent = d;
    if (CLASS_DAYS.includes(ds)) e.classList.add('has-class');
    if (today.getFullYear()===curYear && today.getMonth()===curMonth && today.getDate()===d) e.classList.add('today');
    if (selDate===ds) e.classList.add('selected');
    e.onclick = () => selectDate(ds);
    c.appendChild(e);
  }
}

function changeMonth(d) {
  curMonth+=d;
  if(curMonth>11){curMonth=0;curYear++;}
  if(curMonth<0){curMonth=11;curYear--;}
  renderCal();
}

function selectDate(ds) {
  selDate=ds; selClass=null; students=[]; selected.clear();
  renderCal(); renderClassList(); showEmpty();
}

function renderClassList() {
  const cl = document.getElementById('class-list');
  const dl = document.getElementById('class-date-label');
  const cc = document.getElementById('classes-count');
  const classes = CLASSES[selDate]||[];

  if (selDate) {
    const [y,m,d]=selDate.split('-');
    const dt=new Date(+y,+m-1,+d);
    dl.textContent = DAYS_BR[dt.getDay()] + ', ' + d+'/'+m;
    dl.style.display='block';
    cc.textContent = classes.length ? classes.length+' aula'+(classes.length>1?'s':'') : '';
  }

  if (!classes.length) {
    cl.innerHTML='<div style="text-align:center;padding:20px 0;color:var(--text3);font-size:12px;">Nenhuma aula nesta data</div>';
    return;
  }
  cl.innerHTML = classes.map(c=>`
    <div class="class-card ${selClass&&selClass.id===c.id?'active':''}" onclick="selectClass(${c.id})">
      <div class="class-name">${c.name}</div>
      <div class="class-meta">
        <span>&#128336; ${c.time}</span>
        <span>&#9201; ${c.dur}</span>
        <span>&#128205; ${c.room}</span>
      </div>
      <div class="checkin-badge">${c.checkins}/${c.total}</div>
    </div>
  `).join('');
}

function selectClass(id) {
  const all = Object.values(CLASSES).flat();
  selClass = all.find(c=>c.id===id);
  students = JSON.parse(JSON.stringify(STUDENTS[id]||[]));
  selected.clear(); filter='all'; searchQ='';
  renderClassList(); renderDetail();
  hideBanner();
}

function showEmpty() {
  document.getElementById('empty-state').style.display='flex';
  document.getElementById('class-detail').style.display='none';
}

function renderDetail() {
  document.getElementById('empty-state').style.display='none';
  const det = document.getElementById('class-detail');
  det.style.display='flex';

  const total=students.length;
  const present=students.filter(s=>s.status==='present').length;
  const absent=students.filter(s=>s.status==='absent').length;
  const pending=students.filter(s=>s.status==='pending').length;
  const withCI=students.filter(s=>s.checkin).length;

  let filtered = students.filter(s=>{
    const ms=!searchQ||s.name.toLowerCase().includes(searchQ.toLowerCase());
    const mf=filter==='all'||s.status===filter||(filter==='checkin'&&s.checkin);
    return ms&&mf;
  });

  const [y,m,d]=selDate.split('-');

  // Group for display
  const withCheckin = filtered.filter(s=>s.checkin);
  const withoutCheckin = filtered.filter(s=>!s.checkin);

  det.innerHTML = `
    <div class="class-detail-header">
      <div>
        <div class="class-detail-name">${selClass.name}</div>
        <div class="class-detail-meta">
          <span>&#128197; ${d}/${m}/${y}</span>
          <span>&#128336; ${selClass.time} &middot; ${selClass.dur}</span>
          <span>&#128205; ${selClass.room}</span>
        </div>
      </div>
      <div class="stats-row">
        <div class="stat-chip s-total"><div class="s-num">${total}</div><div class="s-label">Alunos</div></div>
        <div class="stat-chip s-present"><div class="s-num">${present}</div><div class="s-label">Presentes</div></div>
        <div class="stat-chip s-absent"><div class="s-num">${absent}</div><div class="s-label">Faltas</div></div>
        <div class="stat-chip s-pending"><div class="s-num">${pending}</div><div class="s-label">Pendentes</div></div>
      </div>
    </div>

    <div class="bulk-bar">
      <div class="bulk-left">
        <div class="search-wrap">
          <span class="search-icon">&#128269;</span>
          <input class="search-input" placeholder="Buscar aluno..." value="${searchQ}"
            oninput="searchQ=this.value;renderDetail()" />
        </div>
        <div class="filter-tabs">
          <button class="filter-tab ${filter==='all'?'active':''}" onclick="setFilter('all')">Todos (${total})</button>
          <button class="filter-tab ${filter==='checkin'?'active':''}" onclick="setFilter('checkin')">Check-in (${withCI})</button>
          <button class="filter-tab ${filter==='present'?'active':''}" onclick="setFilter('present')">&#10003; (${present})</button>
          <button class="filter-tab ${filter==='absent'?'active':''}" onclick="setFilter('absent')">&#10005; (${absent})</button>
          <button class="filter-tab ${filter==='pending'?'active':''}" onclick="setFilter('pending')">&mdash; (${pending})</button>
        </div>
      </div>
      <div class="bulk-actions">
        ${selected.size>0 ? `
          <span style="font-size:12px;color:var(--text2);">${selected.size} sel.</span>
          <button class="btn btn-present" onclick="bulkSet('present')">&#10003; Presença</button>
          <button class="btn btn-absent" onclick="bulkSet('absent')">&#10005; Falta</button>
          <button class="btn btn-ghost" onclick="selected.clear();renderDetail()">Limpar</button>
        ` : `
          <button class="btn btn-ghost" onclick="selectAll()">Selecionar todos</button>
          <button class="btn btn-present" onclick="markCheckins()">&#10003; Confirmar check-ins</button>
        `}
      </div>
    </div>

    <div class="student-list">
      ${withCheckin.length>0 && filter!=='pending' ? `
        <div class="s-section-label">&#9989; Com check-in (${withCheckin.length})</div>
        ${withCheckin.map(s=>rowHTML(s)).join('')}
      ` : ''}
      ${withoutCheckin.length>0 ? `
        <div class="s-section-label">&#128686; Sem check-in (${withoutCheckin.length})</div>
        ${withoutCheckin.map(s=>rowHTML(s)).join('')}
      ` : ''}
      ${filtered.length===0 ? `<div style="text-align:center;padding:30px;color:var(--text3);font-size:13px;">Nenhum aluno encontrado</div>` : ''}
    </div>
  `;
}

function rowHTML(s) {
  const init = s.name.split(' ').map(n=>n[0]).slice(0,2).join('');
  const bc = BELT_COLORS[s.belt]||'#6b7280';
  const isSel = selected.has(s.id);
  const rowClass = s.status==='present'?'row-present':s.status==='absent'?'row-absent':'';
  return `
    <div class="student-row ${isSel?'sel':''} ${rowClass}" onclick="toggleSel(${s.id})">
      <div class="s-check">${isSel?'&#10003;':''}</div>
      <div class="s-avatar" style="border-color:${bc};color:${bc}">${init}</div>
      <div class="s-info">
        <div class="s-name">${s.name}</div>
        <div class="s-sub">
          <span style="display:inline-block;width:8px;height:8px;background:${bc};border-radius:50%;"></span>
          Faixa ${s.belt}
          ${s.checkin
            ? `<span style="color:var(--present);font-size:10px;">&#9679; Check-in ${s.checkin}</span>`
            : `<span style="color:var(--text3);font-size:10px;">&#9679; Sem check-in</span>`}
        </div>
      </div>
      <div class="s-checkin-time" style="font-size:11px;color:var(--text2);">
        ${s.checkin ? '&#9200; '+s.checkin : ''}
      </div>
      <div class="att-toggle" onclick="event.stopPropagation()">
        <button class="att-btn p ${s.status==='present'?'on':''}" onclick="setSt(${s.id},'present')">&#10003;</button>
        <button class="att-btn a ${s.status==='absent'?'on':''}" onclick="setSt(${s.id},'absent')">&#10005;</button>
      </div>
      <div class="s-status ${s.status}">
        ${s.status==='present'?'Presente':s.status==='absent'?'Falta':'Pendente'}
      </div>
    </div>
  `;
}

function setFilter(f){filter=f;renderDetail();}

function toggleSel(id){
  if(selected.has(id)) selected.delete(id); else selected.add(id);
  renderDetail();
}
function selectAll(){
  students.forEach(s=>selected.add(s.id)); renderDetail();
}
function setSt(id,st){
  const s=students.find(x=>x.id===id);
  if(s){ s.status=s.status===st?'pending':st; markChanged(); renderDetail(); }
}
function bulkSet(st){
  selected.forEach(id=>{const s=students.find(x=>x.id===id);if(s)s.status=st;});
  selected.clear(); markChanged(); renderDetail();
}
function markCheckins(){
  students.filter(s=>s.checkin).forEach(s=>s.status='present');
  markChanged(); renderDetail();
}
function markChanged(){
  const n=students.filter(s=>s.status!=='pending').length;
  document.getElementById('changed-count').textContent=n;
  document.getElementById('save-banner').classList.add('show');
}
function hideBanner(){ document.getElementById('save-banner').classList.remove('show'); }
function saveAttendance(){
  hideBanner();
  const t=document.getElementById('toast');
  const present=students.filter(s=>s.status==='present').length;
  const absent=students.filter(s=>s.status==='absent').length;
  document.getElementById('toast-msg').textContent=`Salvo! ${present} presentes, ${absent} faltas.`;
  t.classList.add('show');
  setTimeout(()=>t.classList.remove('show'),3500);
}
